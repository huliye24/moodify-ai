## 0. 题目定义

## 1. Coverage Index

```python
def compute_coverage_index(history: list) -> dict:
    n = len(history)
    if n == 0:
        return {"coverage": 0.0, "verdict": "insufficient_data"}

    emotion_coverage = len(set(h.emotion_code for h in history)) / 8
    defect_coverage = count_defect_types(history, min_samples=5) / 17
    style_coverage = len(set(h.style for h in history if h.style)) / max(1, len(KNOWN_STYLES))

    coverage = (emotion_coverage * defect_coverage * style_coverage) ** (1/3)
    verdict = "ready" if coverage >= 0.5 else "insufficient"
    return {"coverage": coverage, "verdict": verdict,
            "emotion": emotion_coverage, "defect": defect_coverage, "style": style_coverage}
```

## 2. Label Quality Index

```python
def compute_label_quality(history: list) -> dict:
    labeled = [h for h in history if h.satisfied is not None]
    if len(labeled) < 20:
        return {"quality": 0.0, "verdict": "insufficient_labels"}

    precision = compute_precision(labeled)
    recall = compute_recall(labeled)
    consistency = compute_consistency(labeled)  # 需要重复标注数据

    quality = (precision * recall * consistency) ** (1/3)
    verdict = "reliable" if quality >= 0.7 else "noisy"
    return {"quality": quality, "verdict": verdict}
```

## 3. Teacher Agreement Index

```python
def compute_teacher_agreement(diagnosis, emotion_code, teachers=["deepseek", "search"]):
    recommendations = []
    for teacher in teachers:
        if teacher == "deepseek":
            u = deepseek_client.recommend_params(prompt)
        elif teacher == "search":
            u = search_optimal_strengths(diagnosis, emotion_code)[0][0]

    u_vectors = [strength_vector_to_array(u) for u in recommendations]  # 5D → np.array
    std = np.std(u_vectors, axis=0).mean()
    mean_range = np.ptp(u_vectors, axis=0).mean()

    agreement = 1.0 - std / max(mean_range, 0.01)
    return max(0.0, min(1.0, agreement))
```

## 4. 蒸馏触发决策

```python
def should_distill(history):
    cov = compute_coverage_index(history)
    lab = compute_label_quality(history)
    # Teacher agreement 需要在线计算, 此处用缓存的平均值
    tag = np.mean([h.teacher_agreement for h in history if h.teacher_agreement])

    if cov["coverage"] >= 0.5 and lab["quality"] >= 0.7 and tag >= 0.7:
        return {"distill": True, "reason": "All thresholds met"}
    return {"distill": False, "gaps": {
        "coverage_gap": max(0, 0.5 - cov["coverage"]),
        "quality_gap": max(0, 0.7 - lab["quality"]),
        "agreement_gap": max(0, 0.7 - tag),
    }}
```
