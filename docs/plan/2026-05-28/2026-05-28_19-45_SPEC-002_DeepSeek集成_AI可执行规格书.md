### 2.1 `optimizer/params_to_strengths.py`

**函数签名与实现**:
```python
"""15 参数 → 5 个 transfer function 强度值"""

def params_to_strengths(params: dict[str, float]) -> dict[str, float]:
    p02 = abs(params.get("P02_vocal_presence_gain", 0.0))
    p05 = abs(params.get("P05_proximity_low_gain", 0.0))
    p15 = abs(params.get("P15_high_shelf_gain", 0.0))
    p06 = params.get("P06_compression_ratio", 1.0)
    p11 = params.get("P11_reverb_dry_wet", 0.0)
    p13 = params.get("P13_harmonic_drive", 0.0)

    return {
        "spectrum": min(1.0, (p02 + p05 + p15) / 6.0),
        "dynamic":  min(1.0, (p06 - 1.0) / 3.0),
        "space":    min(1.0, p11 / 0.4),
        "layer":    min(1.0, p13 / 0.3),
        "master":   0.5,
    }
```

**验收**: `get_recommended_params("GA")` 传入 → 返回 5 个 float ∈ [0,1]。

**内部流程** (精确到变量名):
```python
    from moodify.optimizer.params_to_strengths import params_to_strengths
    from moodify.knowledge.emotion_targets import get_ideal_process_vector

    engine = StateTransferEngine()
    ws_raw = StateTransferEngine.diagnostic_to_process(diagnosis)
    target = get_ideal_process_vector(emotion_code)

    strengths = params_to_strengths(params)
    chain_types = ["spectrum", "dynamic", "space", "layer", "master"]
    chain_strengths = [strengths[t] for t in chain_types]
    ws_proxy, meta = engine.apply_chain_transfer(
        ws_raw, chain_types, chain_strengths, emotion_code
    )

    dist_before = float(np.linalg.norm(ws_raw.to_array() - target))
    dist_after  = float(np.linalg.norm(ws_proxy.to_array() - target))

    if dist_before > 1e-6:
        eds = 100.0 * (1.0 - dist_after / dist_before)
    else:
        eds = 100.0

    penalty = 5.0 * len(meta.get("warnings", []))
    return float(np.clip(eds - penalty, -50.0, 100.0))
```

**验收**: 传入诊断 + 推荐参数 → 返回 float ∈ [-50, 100]。

```python
def _run_param_search(self, ws_diagnosis, emotion_target: str) -> PhaseResult:
    t0 = time.perf_counter()
    if ws_diagnosis is None:
        return PhaseResult(phase=1.5, name="参数搜索", status=PhaseStatus.SKIPPED,
                          output={"top_params": [], "top_scores": []}, elapsed_ms=0)
    try:
        from moodify.optimizer.search import search_optimal_params
        results = search_optimal_params(ws_diagnosis, emotion_target, top_k=3)
        return PhaseResult(
            phase=1.5, name="参数搜索", status=PhaseStatus.COMPLETED,
            output={
                "top_params": [r[0] for r in results],
                "top_scores": [r[1] for r in results],
            },
            elapsed_ms=(time.perf_counter() - t0) * 1000,
        )
    except Exception as e:
        import traceback; traceback.print_exc()
        return PhaseResult(
            phase=1.5, name="参数搜索", status=PhaseStatus.COMPLETED,
            output={"top_params": [], "top_scores": [], "fallback_reason": str(e)},
            warnings=[f"Search fallback: {e}"],
            elapsed_ms=(time.perf_counter() - t0) * 1000,
        )
```

```python
"""DeepSeek API 客户端。失败返回 None, 不抛异常。"""
import os, json
from openai import OpenAI

class DeepSeekClient:

    def __init__(self):
        key = os.getenv("DEEPSEEK_API_KEY")
        self._client = OpenAI(
            api_key=key,
            base_url=os.getenv("DEEPSEEK_BASE_URL", "https://api.deepseek.com")
        ) if key else None
        self.model = os.getenv("DEEPSEEK_MODEL", "deepseek-chat")

    @property
    def available(self) -> bool:
        return self._client is not None

    def define_search_space(
        self, diagnosis_dict: dict, defects_list: list[dict],
        emotion_code: str, emotion_name: str
    ) -> dict | None:
        """输入: 诊断 + 缺陷 + 情绪。输出: SearchSpaceResponse dict。失败返回 None。"""
        from moodify.llm.prompts import SEARCH_SPACE_DESIGNER_PROMPT
        user = json.dumps({
            "diagnosis": diagnosis_dict,
            "defects": defects_list,
            "emotion_code": emotion_code,
            "emotion_name": emotion_name,
        }, ensure_ascii=False)
        return self._call(SEARCH_SPACE_DESIGNER_PROMPT, user)

    def narrate_diagnosis(
        self, before_dict: dict, after_dict: dict, params: dict,
        whs_before: float, whs_after: float, eds: float, emotion_name: str
    ) -> dict | None:
        from moodify.llm.prompts import DIAGNOSIS_NARRATOR_PROMPT
        user = json.dumps({
            "diagnosis_before": before_dict,
            "diagnosis_after": after_dict,
            "params_applied": params,
            "whs_before": whs_before, "whs_after": whs_after,
            "eds": eds, "emotion_name": emotion_name,
        }, ensure_ascii=False)
        return self._call(DIAGNOSIS_NARRATOR_PROMPT, user)

    def _call(self, system_prompt: str, user_content: str) -> dict | None:
        if not self.available:
            return None
        for attempt in range(3):
            try:
                r = self._client.chat.completions.create(
                    model=self.model,
                    messages=[
                        {"role": "system", "content": system_prompt},
                        {"role": "user", "content": user_content},
                    ],
                    response_format={"type": "json_object"},
                    timeout=15.0,
                    temperature=0.3,
                )
                return json.loads(r.choices[0].message.content)
            except Exception:
                if attempt == 2:
                    return None
        return None
```

### 4.1 `memory/history.py`

```python
"""处理记录存储与相似案例检索"""
import json, numpy as np
from dataclasses import dataclass, asdict
from pathlib import Path

@dataclass
class ProcessingRecord:
    diagnosis_vector: list[float]
    params: dict[str, float]
    whs_before: float
    whs_after: float
    eds: float
    edsr_proxy: float
    edsr_real: float | None    # None = 无用户反馈
    emotion_code: str
    timestamp: str

class ProcessingHistory:

    def __init__(self, storage_dir: str = "outputs"):
        self._path = Path(storage_dir) / "processing_history.jsonl"

    def save(self, record: ProcessingRecord) -> None:
        self._path.parent.mkdir(parents=True, exist_ok=True)
        with open(self._path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(record), ensure_ascii=False) + "\n")

    def load_all(self) -> list[ProcessingRecord]:
        if not self._path.exists():
            return []
        records = []
        with open(self._path, "r", encoding="utf-8") as f:
            for line in f:
                try:
                    records.append(ProcessingRecord(**json.loads(line)))
                except Exception:
                    pass
        return records

    def find_similar(
        self, query_vector: list[float], top_k: int = 5
    ) -> list[tuple[ProcessingRecord, float]]:
        records = self.load_all()
        if not records:
            return []
        q = np.array(query_vector, dtype=np.float64)
        scored = []
        for r in records:
            v = np.array(r.diagnosis_vector, dtype=np.float64)
            sim = float(np.dot(q, v) / (np.linalg.norm(q) * np.linalg.norm(v) + 1e-8))
            scored.append((r, sim))
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored[:top_k]

def diagnosis_to_vector(diagnosis) -> list[float]:
    """WaveStateDiagnosis → L2 归一化向量 (用于相似度检索)"""
    d = diagnosis.to_dict()
    vec = []
    for dim in ["Spectrum", "Dynamics", "Space", "Layers", "Emotion"]:
        for v in d[dim].values():
            vec.append(1.0 if isinstance(v, bool) and v else
                       0.0 if isinstance(v, bool) else float(v))
    arr = np.array(vec, dtype=np.float64)
    norm = np.linalg.norm(arr)
    return (arr / norm).tolist() if norm > 1e-8 else arr.tolist()
```
