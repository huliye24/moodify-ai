### 2.3 `llm/client.py`

```python
"""DeepSeek API 客户端。失败返回 None, 不抛异常。"""
import os, json
from openai import OpenAI

class DeepSeekClient:

    def __init__(self):
        key = os.getenv("DEEPSEEK_API_KEY")
        self._client = OpenAI(
            api_key=key,
            base_url=os.getenv("DEEPSEEK_BASE_URL", "https://api.deepseek.com")
        ) if key else None
        self.model = os.getenv("DEEPSEEK_MODEL", "deepseek-chat")

    @property
    def available(self) -> bool:
        return self._client is not None

    def recommend_params(self, prompt: str) -> dict | None:
        """输入: 完整 prompt (系统+诊断+案例+意图)。输出: ProcessingRecommendationResponse dict。
        失败返回 None。"""
        from moodify.llm.prompts import PARAM_RECOMMENDER_SYSTEM_PROMPT
        return self._call(PARAM_RECOMMENDER_SYSTEM_PROMPT, prompt)

    def narrate_diagnosis(self, prompt: str) -> dict | None:
        """输入: 诊断前后对比数据。输出: DiagnosisNarrative dict。失败返回 None。"""
        from moodify.llm.prompts import DIAGNOSIS_NARRATOR_PROMPT
        return self._call(DIAGNOSIS_NARRATOR_PROMPT, prompt)

    def _call(self, system: str, user: str) -> dict | None:
        if not self.available:
            return None
        for attempt in range(3):
            try:
                r = self._client.chat.completions.create(
                    model=self.model,
                    messages=[{"role":"system","content":system},{"role":"user","content":user}],
                    response_format={"type":"json_object"},
                    timeout=15.0, temperature=0.3,
                )
                return json.loads(r.choices[0].message.content)
            except Exception:
                if attempt == 2:
                    return None
        return None
```

### 2.4 `llm/prompt_assembler.py`

```python
"""组装发给 DeepSeek 的完整 prompt: 诊断 + 案例 + 意图。"""
import json

def assemble_param_prompt(
    diagnosis_dict: dict,
    defects_list: list[dict],
    emotion_name: str,
    user_intent: str,
    similar_cases: list[dict],  # [{"similarity":0.92, "diagnosis":{...}, "params":{...}, "whs_delta":9, "feedback":"满意"}, ...]
) -> str:
    """组装参数推荐的完整用户 prompt。
    
    Args:
        diagnosis_dict: WaveStateDiagnosis.to_dict()
        defects_list: [{"parameter":"S1_SubPresence","severity":3}, ...]
        emotion_name: "温柔觉醒"
        user_intent: "像深夜的温暖拥抱" 或 "" (自动推断)
        similar_cases: 检索到的相似案例列表, 可为空

    Returns:
        完整的用户 prompt 字符串, ~1500-3000 字符
    """
    parts = []

    # 1. 当前诊断
    parts.append("## 当前诊断\n")
    parts.append("```json\n" + json.dumps(diagnosis_dict, ensure_ascii=False, indent=2) + "\n```\n")

    # 2. 缺陷
    if defects_list:
        parts.append("## 检测到的缺陷\n")
        for d in defects_list:
            parts.append(f"- {d['parameter']}: severity={d['severity']}\n")

    # 3. 目标情绪
    parts.append(f"\n## 目标情绪\n{emotion_name}\n")

    # 4. 用户意图
    if user_intent:
        parts.append(f"\n## 用户意图\n{user_intent}\n")
    else:
        parts.append(f"\n## 用户意图\n自动推断: 适合{emotion_name}方向的处理\n")

    # 5. 相似案例 (核心)
    if similar_cases:
        parts.append(f"\n## 相似案例 (共{len(similar_cases)}个, 来自本地知识库)\n")
        for i, case in enumerate(similar_cases):
            sim = case.get("similarity", 0)
            params = case.get("params", {})
            feedback = case.get("feedback", "")
            whs_d = case.get("whs_delta", 0)
            satisfied = case.get("satisfied", True)

            parts.append(f"### 案例 {i+1} (相似度: {sim:.2f}, 用户{'满意' if satisfied else '不满意'})\n")
            parts.append(f"WHS变化: +{whs_d}\n")
            if feedback:
                parts.append(f"用户反馈: {feedback}\n")

            # 只展示非默认参数 (值偏离 rec 的参数)
            non_default = {k: v for k, v in params.items() if abs(v) > 0.01}
            if non_default:
                parts.append("使用的参数:\n")
                for k, v in non_default.items():
                    parts.append(f"  {k} = {v}\n")
            parts.append("\n")
    else:
        parts.append("\n## 相似案例\n无 (知识库为空, 请基于诊断数据做零样本推理)\n")

    # 6. 输出要求
    parts.append("请基于以上信息, 输出你的参数推荐。必须是严格的 JSON 格式。\n")

    return "".join(parts)


def assemble_narrative_prompt(
    before_dict: dict, after_dict: dict,
    params: dict,
    whs_before: float, whs_after: float, eds: float,
    emotion_name: str,
) -> str:
    """组装诊断解读的 prompt。"""
    return json.dumps({
        "diagnosis_before": before_dict,
        "diagnosis_after": after_dict,
        "params_applied": params,
        "whs_before": whs_before,
        "whs_after": whs_after,
        "eds": eds,
        "emotion_name": emotion_name,
    }, ensure_ascii=False, indent=2)
```

### 2.5 `llm/offline_fallback.py`

```python
"""离线降级: API 不可用时用加权平均聚合相似案例参数。"""
from moodify.knowledge.craft_chains import get_recommended_params

PARAM_KEYS_15 = [
    "P01_vocal_presence_freq", "P02_vocal_presence_gain", "P03_vocal_presence_q",
    "P04_proximity_low_freq", "P05_proximity_low_gain",
    "P06_compression_ratio", "P07_compression_attack", "P08_compression_release", "P09_compression_threshold",
    "P10_reverb_t60", "P11_reverb_dry_wet", "P12_reverb_width",
    "P13_harmonic_drive",
    "P14_high_shelf_freq", "P15_high_shelf_gain",
]

def aggregate_params_offline(
    similar_cases: list[dict],
    emotion_code: str,
) -> dict[str, float]:
    """加权平均聚合相似案例的参数。相似度越高权重越大。用户不满意的案例权重=0。

    Args:
        similar_cases: [{"similarity":0.92, "params":{...}, "satisfied":True}, ...]
        emotion_code: "GA"

    Returns:
        15 参数完整 dict
    """
    if not similar_cases:
        return get_recommended_params(emotion_code)

    # 过滤: 排除不满意案例
    valid = [c for c in similar_cases if c.get("satisfied", True)]
    if not valid:
        return get_recommended_params(emotion_code)

    total_weight = sum(c["similarity"] for c in valid)
    result = {}

    for key in PARAM_KEYS_15:
        weighted_sum = 0.0
        for case in valid:
            val = case.get("params", {}).get(key, 0.0)
            weighted_sum += val * case["similarity"]
        result[key] = weighted_sum / total_weight

    return result
```

    cases = []

修改 `find_similar` 方法, 增加时间衰减:

```python
def find_similar(
    self, query_vector: list[float], top_k: int = 5
) -> list[tuple["ProcessingRecord", float]]:
    """余弦相似度 + 时间衰减。半衰期 180 天。"""
    import math
    from datetime import datetime

    records = self.load_all()
    if not records:
        return []

    now = datetime.now()
    q = np.array(query_vector, dtype=np.float64)
    scored = []
    for r in records:
        # 跳过超 365 天的记录
        try:
            age_days = (now - datetime.fromisoformat(r.timestamp)).days
        except Exception:
            age_days = 0
        if age_days > 365:
            continue

        v = np.array(r.diagnosis_vector, dtype=np.float64)
        cos_sim = float(np.dot(q, v) / (np.linalg.norm(q) * np.linalg.norm(v) + 1e-8))
        time_decay = math.exp(-age_days / 180.0)
        final_sim = cos_sim * time_decay
        scored.append((r, final_sim))

    scored.sort(key=lambda x: x[1], reverse=True)
    return scored[:top_k]
```

### 3.2 `optimizer/search.py` — `define_search_space` 改为 3 层

保留原有函数, 新增:

```python
def recommend_params_with_rag(
    diagnosis,
    emotion_target: str,
    user_intent: str = "",
    history: "ProcessingHistory | None" = None,
) -> tuple[dict[str, float], float, str]:
    """3层决策架构: RAG + LLM → LLM零样本 → 搜索回退。
    
    Returns:
        (params_dict, confidence, source)
        source ∈ {"rag_llm", "llm_zeroshot", "search", "offline_aggregation", "preset"}
    """
    from moodify.knowledge.emotion_targets import resolve_emotion, EMOTION_TARGETS_V2
    from moodify.diagnosis.defect_classifier import DefectClassifier

    emotion_code = resolve_emotion(emotion_target, as_key=True)
    emotion_name = EMOTION_TARGETS_V2.get(emotion_code, {}).get("name_zh", emotion_target)

    # 分类缺陷
    classifier = DefectClassifier()
    defects = classifier.classify(diagnosis, emotion_code)
    defects_list = [{"parameter": d.parameter, "severity": d.severity} for d in defects]

    # 检索相似案例
    similar_cases_raw = []
    if history:
        from moodify.memory.history import diagnosis_to_vector
        dvec = diagnosis_to_vector(diagnosis)
        similar_cases_raw = history.find_similar(dvec, top_k=5)

    top1_sim = similar_cases_raw[0][1] if similar_cases_raw else 0.0

    # ===== Layer 1: 案例充足 → LLM 直接推理 =====
    if top1_sim > 0.85:
        try:
            from moodify.llm.client import DeepSeekClient
            from moodify.llm.prompt_assembler import assemble_param_prompt

            client = DeepSeekClient()
            if client.available and history:
                cases = history.format_cases_for_prompt(similar_cases_raw)
                prompt = assemble_param_prompt(
                    diagnosis.to_dict(), defects_list, emotion_name, user_intent, cases
                )
                result = client.recommend_params(prompt)
                if result and "parameters" in result:
                    params = {p["param_name"]: p["value"] for p in result["parameters"]}
                    confidence = result.get("confidence", 0.8)
                    return params, confidence, "rag_llm"
        except Exception:
            pass

    # ===== Layer 2: LLM 零样本 (无案例或案例不够相似) =====
    if top1_sim < 0.85:
        try:
            from moodify.llm.client import DeepSeekClient
            from moodify.llm.prompt_assembler import assemble_param_prompt

            client = DeepSeekClient()
            if client.available:
                # 可能有案例但不够相似, 仍然传入作为弱参考
                cases = history.format_cases_for_prompt(similar_cases_raw) if history else []
                prompt = assemble_param_prompt(
                    diagnosis.to_dict(), defects_list, emotion_name, user_intent, cases
                )
                result = client.recommend_params(prompt)
                if result and "parameters" in result:
                    params = {p["param_name"]: p["value"] for p in result["parameters"]}
                    confidence = result.get("confidence", 0.6)
                    return params, confidence, "llm_zeroshot"
        except Exception:
            pass

    # ===== Layer 3: 离线降级 (无 API) =====
    if similar_cases_raw and history:
        from moodify.llm.offline_fallback import aggregate_params_offline
        cases = history.format_cases_for_prompt(similar_cases_raw)
        params = aggregate_params_offline(cases, emotion_code)
        return params, 0.5, "offline_aggregation"

    # ===== 最终回退: 搜索 =====
    try:
        results = search_optimal_params(diagnosis, emotion_target, top_k=1)
        if results:
            return results[0][0], 0.4, "search"
    except Exception:
        pass

    # ===== 兜底: 预设 =====
    from moodify.knowledge.craft_chains import get_recommended_params
    return get_recommended_params(emotion_code), 0.3, "preset"
```

## 4. 删除: Phase 1.5 多版本搜索逻辑

SPEC-002 的 Phase 3 多版本循环 + Phase 4-6 循环选最优的逻辑**不再需要**。原因是:

1. Layer 1 (案例充足): LLM 直接给出高置信度参数, 不需要多版本比较
2. Layer 2 (零样本): LLM 的参数质量足够, proxy 验证即可, 不需要 top-3 真实 DSP
3. Layer 3 (离线/回退): 已有加权平均或预设, 不需要搜索

**多版本 DSP 只在主动 A/B 测试模式时启用 (用户选择"比较多个方案")。** 默认单版本, 速度快 3 倍。

## 7. 关键设计决策 (为什么这样做)

1. **单版本处理, 非多版本** — LLM + RAG 输出已足够精确, 不需要 top-3 DSP 选最优。节省 3 倍处理时间。

2. **3 层决策, 非硬分支** — 不是「有 API 用 API / 没 API 用搜索」。而是根据案例充足度动态选择推理路径。案例越丰富, LLM 越直接。

3. **离线不是"降级体验", 是"替代体验"** — 加权平均聚合不是"劣化版 LLM", 是统计学上合理的推荐方式。相似案例的加权平均天然接近最优。

4. **历史案例进入 prompt, 不是事后校准** — 这是云景台方案对 SPEC-002 的核心修正。检索在 LLM 调用之前, 不在之后。

5. **用户自带 API Key (BYOK)** — Moodify 不承担 API 费用。未配置 Key 时自动使用离线聚合。两种模式都可用。
